async function start(){ try { if ( !cfg.SUPABASE_URL || !cfg.SUPABASE_PUBLISHABLE_KEY || !cfg.LIFF_ID || cfg.LIFF_ID === 'YOUR_LIFF_ID' ) { throw new Error('會員中心尚未設定完成，請聯絡 NAVO。'); } await liff.init({ liffId: cfg.LIFF_ID }); // LINE 內建瀏覽器會由 LIFF 初始化處理登入， // 不要在這裡重複呼叫 liff.login() if (!liff.isInClient() && !liff.isLoggedIn()) { liff.login({ redirectUri: location.href }); return; } idToken = liff.getIDToken(); if (!idToken) { throw new Error( '沒有取得 LINE 登入憑證，請確認 LIFF 已啟用 openid 權限，並重新開啟會員頁面。' ); } await loadProfile(); } catch (e) { show('loading'); message( 'loading', '會員中心開啟失敗：' + (e.message || '未知錯誤') + '\n請關閉頁面後重新開啟；若仍失敗，請聯絡 NAVO。' ); } }
